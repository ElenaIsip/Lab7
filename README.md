# Разбор по шагам:
Объявление переменной char c — для хранения одного символа.

Вывод приглашения printf("Введите символы 'a' or 'b': ").

Считывание символа с консоли через scanf("%c", &c).

Конструкция switch (c) проверяет значение c:

* если c == 'a' → выводится "Введено 'a'.", затем break (выход из switch);

* если c == 'b' → выводится "Введено 'b'.", затем break;

иначе (любой другой символ) → срабатывает default → выводится "Неизвестный символ".

return 0; — завершение программы.

# Особенности работы
scanf("%c", ...) читает ровно один символ, включая пробел, табуляцию и перевод строки (\n).

Если пользователь нажмёт Enter, не введя символ, то будет прочитан '\n', и сработает ветка default → "Неизвестный символ".

Регистр важен: 'A' и 'B' не совпадут с 'a' и 'b', поэтому тоже попадут в default.

Если ввести строку из нескольких символов (например, abc), будет прочитан только первый символ 'a' — остальные останутся в буфере ввода.

# Код программы




    int n;   


    printf("Введите показатель степени 10 (например, -9, 3, 6): ");
    if (scanf("%d", &n) != 1) {
        printf("Ошибка: введите целое число!\n");
        return 1;
    }

    printf("10^%d — ", n);

    switch (n) {
    case -24: printf("иокто (и)\n");    break;
    case -21: printf("зепто (з)\n");    break;
    case -18: printf("атто (а)\n");     break;
    case -15: printf("фемто (ф)\n");    break;
    case -12: printf("пико (п)\n");     break;
    case  -9: printf("нано (н)\n");     break;
    case  -6: printf("микро (мк)\n");   break;
    case  -3: printf("милли (м)\n");    break;
    case  -2: printf("санти (с)\n");    break;
    case  -1: printf("деци (д)\n");     break;
    case   0: printf("(без приставки)\n"); break;
    case   1: printf("дека (да)\n");    break;
    case   2: printf("гекто (г)\n");    break;
    case   3: printf("кило (к)\n");     break;
    case   6: printf("мега (М)\n");     break;
    case   9: printf("гига (Г)\n");     break;
    case  12: printf("тера (Т)\n");     break;
    case  15: printf("пета (П)\n");     break;
    case  18: printf("экса (Э)\n");     break;
    case  21: printf("зетта (З)\n");    break;
    case  24: printf("йотта (И)\n");    break;
    default:
        printf("нет соответствующей приставки СИ\n");
    }

    return 0;


 
# Вывод
Программа реализует многовариантный выбор (switch с 21 ветвью + default) и дополнительно проверку корректности ввода. Алгоритм:

* линеен до момента ввода,

* далее содержит развилку (проверка scanf),

* затем многоветвевую структуру (switch),

*  завершается возвратом кода (0 при успехе, 1 при ошибке ввода).

Такой подход типичен для задач классификации: по числовому коду (n) выбирается одна из заранее известных текстовых категорий (приставка СИ), а «неизвестные» коды обрабатываются веткой default.
